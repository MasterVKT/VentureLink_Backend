Documentation des Services Firebase VentureLink
Ce document détaille l'intégration des services Firebase dans l'application VentureLink, fournissant une référence commune pour les équipes frontend et backend afin d'assurer une intégration cohérente des fonctionnalités Firebase.
1. Vue d'ensemble
VentureLink utilise plusieurs services Firebase pour offrir des fonctionnalités essentielles:
•	Firebase Authentication: Gestion de l'authentification utilisateur
•	Firebase Cloud Messaging (FCM): Notifications push
•	Firebase Analytics (recommandé): Suivi de l'engagement utilisateur
•	Firebase Crashlytics (recommandé): Suivi des crashes

Future<UserCredential> registerWithEmailPassword(String email, String password) async {
  try {
    return await FirebaseAuth.instance.createUserWithEmailAndPassword(
      email: email,
      password: password,
    );
  } on FirebaseAuthException catch (e) {
    // Gérer les erreurs spécifiques
    if (e.code == 'email-already-in-use') {
      throw AuthException('EMAIL_IN_USE');
    } else if (e.code == 'weak-password') {
      throw AuthException('WEAK_PASSWORD');
    }
    throw AuthException('REGISTRATION_FAILED');
  }
}

Future<UserCredential> signInWithEmailPassword(String email, String password) async {
  try {
    return await FirebaseAuth.instance.signInWithEmailAndPassword(
      email: email,
      password: password,
    );
  } on FirebaseAuthException catch (e) {
    // Gérer les erreurs
    if (e.code == 'user-not-found' || e.code == 'wrong-password') {
      throw AuthException('INVALID_CREDENTIALS');
    }
    throw AuthException('LOGIN_FAILED');
  }
}

Future<UserCredential> signInWithGoogle() async {
  // Déclencher le flux d'authentification
  final GoogleSignInAccount? googleUser = await GoogleSignIn().signIn();
  
  if (googleUser == null) throw AuthException('SIGN_IN_CANCELLED');
  
  // Obtenir les détails d'authentification
  final GoogleSignInAuthentication googleAuth = await googleUser.authentication;
  
  // Créer un credential Firebase
  final credential = GoogleAuthProvider.credential(
    accessToken: googleAuth.accessToken,
    idToken: googleAuth.idToken,
  );
  
  // Se connecter à Firebase avec ce credential
  return await FirebaseAuth.instance.signInWithCredential(credential);
}

def verify_firebase_token(token):
    try:
        decoded_token = auth.verify_id_token(token)
        return decoded_token
    except auth.InvalidIdTokenError:
        raise AuthenticationError("Invalid token")
    except auth.ExpiredIdTokenError:
        raise AuthenticationError("Token expired")
    except Exception as e:
        raise AuthenticationError(f"Authentication failed: {str(e)}")
Middleware d'authentification
class FirebaseAuthMiddleware:
    def __init__(self, get_response):
        self.get_response = get_response

    def __call__(self, request):
        authorization_header = request.headers.get('Authorization')
        
        if not authorization_header or not authorization_header.startswith('Bearer '):
            return self.unauthorized_response()
        
        token = authorization_header.split('Bearer ')[1]
        
        try:
            decoded_token = auth.verify_id_token(token)
            request.firebase_user = decoded_token
        except Exception:
            return self.unauthorized_response()
        
        return self.get_response(request)
    
    def unauthorized_response(self):
        return JsonResponse({'error': 'Unauthorized'}, status=401)

Initialisation FCM
Future<void> initializeFCM() async {
  // Demander la permission
  NotificationSettings settings = await FirebaseMessaging.instance.requestPermission(
    alert: true,
    badge: true,
    sound: true,
  );
  
  if (settings.authorizationStatus == AuthorizationStatus.authorized) {
    // S'abonner aux topics généraux
    await FirebaseMessaging.instance.subscribeToTopic('announcements');
    
    // Configurer les gestionnaires
    FirebaseMessaging.onMessage.listen(_handleForegroundMessage);
    FirebaseMessaging.onMessageOpenedApp.listen(_handleMessageOpenedApp);
    
    // Configurer les canaux Android
    if (Platform.isAndroid) {
      await _setupAndroidNotificationChannels();
    }
    
    // Obtenir et enregistrer le token
    String? token = await FirebaseMessaging.instance.getToken();
    if (token != null) {
      await _registerFCMToken(token);
    }
    
    // Écouter les changements de token
    FirebaseMessaging.instance.onTokenRefresh.listen(_registerFCMToken);
  }
}

void _handleForegroundMessage(RemoteMessage message) {
  // Traiter les notifications reçues lorsque l'app est au premier plan

  }
}

from firebase_admin import messaging

def send_fcm_notification(token, title, body, data=None):
    """
    Envoie une notification FCM à un token spécifique
    """
    message = messaging.Message(
        notification=messaging.Notification(
            title=title,
            body=body
        ),
        data=data or {},
        token=token,
        android=messaging.AndroidConfig(
            priority='high',
            notification=messaging.AndroidNotification(
                channel_id=_get_channel_id(data.get('type') if data else None)
            )
        ),
        apns=messaging.APNSConfig(
            headers={'apns-priority': '10'},
            payload=messaging.APNSPayload(
                aps=messaging.Aps(
                    badge=1,
                    sound='default'
                )
            )
        )
    )
    
    try:
        response = messaging.send(message)
        return response
    except Exception as e:
        logger.error(f"Failed to send FCM notification: {str(e)}")
        raise NotificationError("Failed to send notification")

def _get_channel_id(notification_type):
    """
    Détermine le canal de notification Android approprié
    """
    if notification_type in ['NEW_MESSAGE']:
        return 'messages'
    elif notification_type in ['NEW_INTEREST', 'INTEREST_ACCEPTED', 'INTEREST_REJECTED']:
        return 'interests'
    elif notification_type in ['PROJECT_QUESTION', 'QUESTION_ANSWERED']:
        return 'questions'
    else:
        return 'general'

def send_fcm_multicast(tokens, title, body, data=None):
    """
    Envoie une notification FCM à plusieurs tokens
    """
    message = messaging.MulticastMessage(
        notification=messaging.Notification(
            title=title,
            body=body
        ),
        data=data or {},
        tokens=tokens,
        android=messaging.AndroidConfig(
            priority='high',
            notification=messaging.AndroidNotification(
                channel_id=_get_channel_id(data.get('type') if data else None)
            )
        ),
        apns=messaging.APNSConfig(
            headers={'apns-priority': '10'},
            payload=messaging.APNSPayload(
                aps=messaging.Aps(
                    badge=1,
                    sound='default'
                )
            )
        )
    )
    
    try:
        response = messaging.send_multicast(message)
        return response
    except Exception as e:
        logger.error(f"Failed to send FCM multicast: {str(e)}")
        raise NotificationError("Failed to send notification")

def send_fcm_topic(topic, title, body, data=None):
    """
    Envoie une notification FCM à un topic
    """
    message = messaging.Message(
        notification=messaging.Notification(
            title=title,
            body=body
        ),
        data=data or {},
        topic=topic,
        android=messaging.AndroidConfig(
            priority='normal',
            notification=messaging.AndroidNotification(
                channel_id='general'
            )
        )
    )
    
    try:
        response = messaging.send(message)
        return response
    except Exception as e:
        logger.error(f"Failed to send FCM topic message: {str(e)}")
        raise NotificationError("Failed to send notification")

def notify_new_interest(interest):
    """
    Notifie un créateur de projet d'une nouvelle marque d'intérêt
    """
    # Récupérer le token FCM du créateur
    creator = interest.project.creator
    token = creator.fcm_token
    
    if not token:
        logger.warning(f"User {creator.id} has no FCM token")
        return
    
    # Créer les données de notification
    title = "Nouvel intérêt pour votre projet"
    body = f"{interest.user.first_name} {interest.user.last_name} est intéressé par votre projet {interest.project.title}"
    data = {
        "type": "NEW_INTEREST",
        "entity_id": str(interest.id),
        "action": "OPEN_PROJECT_INTEREST",
        "project_id": str(interest.project.id),
        "interest_id": str(interest.id)
    }
    
    # Envoyer la notification
    try:
        send_fcm_notification(token, title, body, data)
        
        # Enregistrer la notification dans la base de données
        Notification.objects.create(
            recipient=creator,
            sender=interest.user,
            notification_type="NEW_INTEREST",
            title=title,
            message=body,
            entity_type="PROJECT_INTEREST",
            entity_id=interest.id,
            action_url=f"/projects/{interest.project.id}/interests/{interest.id}",
            delivered=True,
            delivered_at=timezone.now()
        )
    except Exception as e:
        logger.error(f"Failed to notify new interest: {str(e)}")
        
        # Enregistrer quand même la notification dans la base de données
        Notification.objects.create(
            recipient=creator,
            sender=interest.user,
            notification_type="NEW_INTEREST",
            title=title,
            message=body,
            entity_type="PROJECT_INTEREST",
            entity_id=interest.id,
            action_url=f"/projects/{interest.project.id}/interests/{interest.id}",
            delivered=False
        )

        'project_title': projectTitle,
        'category': category,
      },
    );
  }

      parameters: {
        'project_id': projectId,
        'project_title': projectTitle,
      },
    );
  }

    await _analytics.logEvent(

      parameters: {
        'conversation_id': conversationId,
        'has_attachment': hasAttachment,
      },
    );
  }

    required String userId,
    required String userType,
    required bool isPremium,

  // Navigation entre écrans
  Future<void> logScreenView({
    required String screenName,
    String? screenClass,
  }) async {
    await _analytics.logScreenView(
      screenName: screenName,
      screenClass: screenClass,
    );
  }
}

7. Sécurité et bonnes pratiques
7.1 Gestion des tokens JWT
•	Les tokens JWT générés par Firebase Authentication ont une durée de validité de 1 heure
•	Le refresh token a une durée de validité de 2 semaines
•	Le frontend doit toujours vérifier si le token est expiré avant d'envoyer une requête
•	En cas d'expiration, utiliser le refresh token pour obtenir un nouveau token
7.2 Sécurisation des communications
•	Toutes les communications avec Firebase doivent être effectuées via HTTPS
•	Les clés API Firebase doivent être restreintes par origine dans la console Firebase
•	Pour les applications Android, restreindre les clés par empreinte de l'application
•	Pour les applications iOS, restreindre les clés par bundle ID
7.3 Gestion des permissions Firebase
•	Utiliser des règles de sécurité restrictives pour les services Firebase (si utilisés directement)
•	Limiter les permissions du compte de service du SDK Admin au strict minimum
7.4 Monitoring et débuggage
•	Activer la journalisation avancée dans la console Firebase
•	Configurer des alertes pour les événements critiques (échecs d'authentification multiples, etc.)
•	Vérifier régulièrement les rapports d'utilisation pour détecter des anomalies
